cmake_minimum_required(VERSION 3.25)
project(msys_compat VERSION 0.1.0 LANGUAGES CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MSVC AMD64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
# Shared sources are compiled separately under each target's CRT/stack rules.
set(compat_core src/core/identify.cpp src/core/process.cpp src/core/token_context.cpp)
add_library(msys-compat-agent SHARED ${compat_core} src/agent/hooks.cpp src/agent/memory.cpp src/agent/stack_probe.asm)
target_include_directories(msys-compat-agent PRIVATE include)
set_target_properties(msys-compat-agent PROPERTIES OUTPUT_NAME msys-compat
  ARCHIVE_OUTPUT_NAME msys-compat-agent PDB_NAME msys-compat-agent)
target_compile_options(msys-compat-agent PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/GS-> $<$<COMPILE_LANGUAGE:CXX>:/GR-> $<$<COMPILE_LANGUAGE:CXX>:/Zl> $<$<COMPILE_LANGUAGE:CXX>:/Oi-> $<$<COMPILE_LANGUAGE:CXX>:/EHs-c->)
target_link_options(msys-compat-agent PRIVATE /NODEFAULTLIB /NOENTRY /INCREMENTAL:NO /GUARD:NO /DYNAMICBASE)
# Keep the normal probe threshold: do not call __chkstk in every small hook
# frame on MSYS-managed secondary-thread stacks.
target_compile_options(msys-compat-agent PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/Gs4096>)
add_executable(msys-compat ${compat_core} src/host/main.cpp src/host/platform.cpp src/host/security.cpp src/host/bootstrap.cpp)
target_include_directories(msys-compat PRIVATE include)
target_compile_options(msys-compat PRIVATE /EHsc)
target_link_libraries(msys-compat PRIVATE advapi32 userenv)
add_dependencies(msys-compat msys-compat-agent)
option(BUILD_TESTING "Build regression suites and fixtures" OFF)
option(MSYS_COMPAT_BUILD_TOOLS "Build the local PSEC development launcher" OFF)
if(MSYS_COMPAT_BUILD_TOOLS OR BUILD_TESTING)
  add_executable(psec-run tools/psec-run/main.cpp)
  target_include_directories(psec-run PRIVATE include third_party)
  target_compile_options(psec-run PRIVATE /EHsc)
  set_target_properties(psec-run PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(psec-run PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/tools")
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS msys-compat msys-compat-agent RUNTIME DESTINATION bin)
install(FILES README.md CHANGELOG.md DESTINATION share/msys-compat)
install(DIRECTORY docs/ DESTINATION share/msys-compat/docs FILES_MATCHING PATTERN "*.md")
install(FILES third_party/LICENSE.nlohmann third_party/LICENSE.mxc third_party/LICENSE.flatbuffers
  DESTINATION share/msys-compat/licenses)
